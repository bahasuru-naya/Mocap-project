import bpy
import socket
import json

# Settings
UDP_IP = "127.0.0.1"
UDP_PORT = 5052
SCALE = 5.0 

# 1. Setup Socket
if "mocap_sock" not in globals():
    global mocap_sock
    mocap_sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    mocap_sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    mocap_sock.bind((UDP_IP, UDP_PORT))
    mocap_sock.setblocking(False)

# 2. Create or Get Collection for Mocap Points
coll_name = "MediaPipe_Points"
if coll_name not in bpy.data.collections:
    mocap_coll = bpy.data.collections.new(coll_name)
    bpy.context.scene.collection.children.link(mocap_coll)
else:
    mocap_coll = bpy.data.collections[coll_name]

# 3. Create 33 Points (Low-level method to avoid Context Errors)
for i in range(33):
    obj_name = f"MP_Point_{i}"
    if obj_name not in bpy.data.objects:
        # Create a new Empty data block and object
        empty_data = bpy.data.objects.new(obj_name, None)
        empty_data.empty_display_type = 'SPHERE'
        empty_data.empty_display_size = 0.1
        # Link it to our collection
        mocap_coll.objects.link(empty_data)

def update_visualizer():
    try:
        data, addr = mocap_sock.recvfrom(10240)
        if data:
            lm_list = json.loads(data.decode())
            
            for i in range(33):
                obj = bpy.data.objects.get(f"MP_Point_{i}")
                if obj:
                    # Update location directly
                    # Mapping: MP X -> BL X, MP Z -> BL Y, -MP Y -> BL Z
                    obj.location = (
                        lm_list[i][0] * SCALE, 
                        lm_list[i][2] * SCALE, 
                        lm_list[i][1] * SCALE 
                    )
            
            # Force refresh the 3D view
            bpy.context.view_layer.update()
            
    except BlockingIOError:
        pass
    except Exception as e:
        print(f"Update Error: {e}")
        
    return 0.01

# Register Timer
if not bpy.app.timers.is_registered(update_visualizer):
    bpy.app.timers.register(update_visualizer)
    print("Mocap Visualizer Started!")
